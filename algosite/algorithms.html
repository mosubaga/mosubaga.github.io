Push and Pop

Push and pop are the two core operations of a stack. Push adds an item to the top of the stack. Pop removes the item from the top. Because both work only on one end of an underlying array, both are O(1).

Think of it like… a spring-loaded plate dispenser. Push a new plate down (push). Grab the top plate to serve (pop). The bottom plate is always the one pushed first — it never comes out until everything above it is removed.

Step by Step

  1. Push: Append the new value to the end of the underlying array. The array's last element is now the "top."
  2. Pop: Read the value at the last index, shrink the array by one, and return that value.
  3. Peek: Read the value at the last index without shrinking the array.

Code Examples

Python 
# Push = append to end. Pop = remove from end. Both O(1).
stack = []

# --- PUSH ---
stack.append("A")    # stack: ["A"]
stack.append("B")    # stack: ["A", "B"]
stack.append("C")    # stack: ["A", "B", "C"]

# --- PEEK (read top without removing) ---
print(stack[-1])     # "C"

# --- POP ---
print(stack.pop())   # "C"  — stack: ["A", "B"]
print(stack.pop())   # "B"  — stack: ["A"]

# Safety check before popping
if stack:
    print(stack.pop())   # "A"  — stack: []

print(stack)         # [] — empty


# Practical example: matching brackets
def is_balanced(s):
    stack = []
    pairs = {')': '(', ']': '[', '}': '{'}
    for ch in s:
        if ch in "([{":
            stack.append(ch)           # push opening bracket
        elif ch in ")]}":
            if not stack or stack[-1] != pairs[ch]:
                return False
            stack.pop()               # pop matching opener
    return len(stack) == 0            # balanced if nothing left

print(is_balanced("({[]})"))   # True
print(is_balanced("({[})"))    # False
TypeScript 
class Stack<T> {
  private items: T[] = [];

  push(val: T): void    { this.items.push(val); }           // O(1)
  pop(): T | undefined  { return this.items.pop(); }        // O(1)
  peek(): T | undefined { return this.items[this.items.length - 1]; } // O(1)
  isEmpty(): boolean    { return this.items.length === 0; }
  size(): number        { return this.items.length; }
}

const stack = new Stack<string>();
stack.push("A");  // ["A"]
stack.push("B");  // ["A", "B"]
stack.push("C");  // ["A", "B", "C"]

console.log(stack.peek()); // "C"
console.log(stack.pop());  // "C"
console.log(stack.pop());  // "B"
console.log(stack.size()); // 1

// Practical: evaluate reverse Polish notation (e.g. "2 3 + 4 *" = 20)
function evalRPN(tokens: string[]): number {
  const stack = new Stack<number>();
  for (const t of tokens) {
    if (["+", "-", "*", "/"].includes(t)) {
      const b = stack.pop()!, a = stack.pop()!;
      if (t === "+") stack.push(a + b);
      if (t === "-") stack.push(a - b);
      if (t === "*") stack.push(a * b);
      if (t === "/") stack.push(Math.trunc(a / b));
    } else {
      stack.push(Number(t));
    }
  }
  return stack.pop()!;
}

console.log(evalRPN(["2","3","+","4","*"])); // 20
Where you see this in real software: Undo systems, function call stack (every program uses this to track nested function calls), expression evaluation, syntax parsing, and backtracking in search algorithms.

Enqueue and Dequeue

Enqueue and dequeue are the core operations of a queue. Enqueue adds an item to the back. Dequeue removes the item from the front. This enforces First In, First Out (FIFO) ordering — the longest-waiting item is always served next.

Think of it like… a checkout line at a grocery store. New shoppers join at the back (enqueue). The cashier serves the person at the front (dequeue). The first person to arrive is the first to leave — no cutting in line.

Step by Step

  1. Enqueue: Append the new item to the back (tail) of the queue.
  2. Dequeue: Remove and return the item at the front (head) of the queue.
  3. Key concern: A plain array's shift() is O(n) because it moves all elements. A proper queue uses a linked list or circular buffer for O(1) dequeue.

Code Examples

Python 
from collections import deque

# deque = doubly-ended queue — O(1) on both ends
q = deque()

# --- ENQUEUE (add to back) ---
q.append("job1")    # q: deque(["job1"])
q.append("job2")    # q: deque(["job1", "job2"])
q.append("job3")    # q: deque(["job1", "job2", "job3"])

# Peek at front without removing
print(q[0])         # "job1"

# --- DEQUEUE (remove from front) ---
print(q.popleft())  # "job1"  — O(1)
print(q.popleft())  # "job2"  — O(1)
print(list(q))      # ["job3"]


# Practical example: process tasks round-robin
tasks = deque(["task_A", "task_B", "task_C"])
workers = 2

while tasks:
    # Each cycle processes up to `workers` tasks
    for _ in range(min(workers, len(tasks))):
        task = tasks.popleft()
        print(f"Processing: {task}")
    # Could re-enqueue partially-done tasks here
TypeScript 
// O(1) queue using a singly linked list
class QNode<T> {
  constructor(public val: T, public next: QNode<T> | null = null) {}
}

class Queue<T> {
  private head: QNode<T> | null = null;
  private tail: QNode<T> | null = null;
  private _size = 0;

  enqueue(val: T): void {               // O(1) — add to back
    const node = new QNode(val);
    if (this.tail) this.tail.next = node;
    this.tail = node;
    if (!this.head) this.head = node;
    this._size++;
  }

  dequeue(): T | undefined {            // O(1) — remove from front
    if (!this.head) return undefined;
    const val = this.head.val;
    this.head = this.head.next;
    if (!this.head) this.tail = null;
    this._size--;
    return val;
  }

  peek(): T | undefined { return this.head?.val; }
  isEmpty(): boolean    { return this._size === 0; }
  size(): number        { return this._size; }
}

const q = new Queue<string>();
q.enqueue("request_1");
q.enqueue("request_2");
q.enqueue("request_3");

// Web server handles requests in order
while (!q.isEmpty()) {
  console.log("Handling:", q.dequeue());
}
// Handling: request_1
// Handling: request_2
// Handling: request_3
Where you see this in real software: Web server request queues, message brokers (Kafka, SQS), print spoolers, CPU task scheduling, and BFS graph traversal — which relies on a queue to process nodes level by level.

Hashing

Hashing is the process of converting a key (like a string or number) into a fixed-range integer called a hash code, which is then used as an array index (a "bucket"). A good hash function distributes keys uniformly across buckets to minimize collisions. The same key must always produce the same hash — hashing is deterministic.

Think of it like… assigning lockers by name initials. "Alice" → locker A, "Bob" → locker B. The "hash function" is the initials rule. It runs instantly and always gives the same answer. The problem: "Alice" and "Anna" both map to A — that's a collision, which needs separate handling.

Step by Step

  1. Take the key (e.g. the string "cat").
  2. Run it through the hash function: sum the character codes, multiply by a prime, then take modulo table size. Result: a bucket index like 3.
  3. Store the value at bucket 3 in an array.
  4. On lookup: run the same hash function → jump straight to bucket 3 → return the value. No scanning needed.

Code Examples

Python 
# How a hash function works internally (simplified)
def naive_hash(key: str, table_size: int) -> int:
    """Sum char codes, multiply by prime, mod by table size."""
    h = 0
    for ch in key:
        h = (h * 31 + ord(ch)) % table_size
    return h

table_size = 10
print(naive_hash("cat", table_size))    # e.g. 1
print(naive_hash("dog", table_size))    # e.g. 7
print(naive_hash("rat", table_size))    # some bucket

# Python's built-in hash() (more sophisticated, randomized per run)
print(hash("hello"))     # integer (varies per Python session)
print(hash(42))          # 42 (small ints hash to themselves)
print(hash((1, 2, 3)))   # tuples are hashable; lists are not

# A minimal hashmap built on top of a hash function + chaining
class SimpleHashMap:
    def __init__(self, size=16):
        self.size = size
        self.buckets = [[] for _ in range(size)]

    def _bucket(self, key):
        h = 0
        for ch in str(key):
            h = (h * 31 + ord(ch)) % self.size
        return h

    def put(self, key, val):
        b = self.buckets[self._bucket(key)]
        for i, (k, _) in enumerate(b):
            if k == key:
                b[i] = (key, val)
                return
        b.append((key, val))

    def get(self, key):
        for k, v in self.buckets[self._bucket(key)]:
            if k == key:
                return v
        return None

hm = SimpleHashMap()
hm.put("name", "Alice")
hm.put("role", "engineer")
print(hm.get("name"))   # "Alice"
print(hm.get("role"))   # "engineer"
TypeScript 
// A simple polynomial hash function
function polyHash(key: string, tableSize: number): number {
  let h = 0;
  for (const ch of key) {
    h = (Math.imul(h, 31) + ch.charCodeAt(0)) % tableSize;
  }
  return h;
}

console.log(polyHash("cat", 10));  // e.g. 1
console.log(polyHash("dog", 10));  // e.g. 7

// Minimal hashmap with chaining
class HashMap<V> {
  private buckets: Array<Array<[string, V]>>;
  private size: number;

  constructor(size = 16) {
    this.size = size;
    this.buckets = Array.from({ length: size }, () => []);
  }

  private bucket(key: string): number {
    return polyHash(key, this.size);
  }

  set(key: string, val: V): void {
    const b = this.buckets[this.bucket(key)];
    const idx = b.findIndex(([k]) => k === key);
    if (idx >= 0) b[idx] = [key, val];
    else b.push([key, val]);
  }

  get(key: string): V | undefined {
    return this.buckets[this.bucket(key)].find(([k]) => k === key)?.[1];
  }

  has(key: string): boolean {
    return this.get(key) !== undefined;
  }
}

const hm = new HashMap<string>();
hm.set("name", "Alice");
hm.set("role", "engineer");
console.log(hm.get("name")); // "Alice"
console.log(hm.has("age"));  // false
Where you see this in real software: Every language's built-in dictionary/map/object, database index hashing, password storage (cryptographic hashes), checksums, load balancing (consistent hashing), and content-addressable storage (git uses SHA-1 hashes for every commit and file).

Collision Resolution

A collision occurs when two different keys hash to the same bucket. Since hash functions are not perfect, collisions happen even with good hash functions. Two main strategies handle them: chaining (each bucket holds a list of items) and open addressing (probe other buckets for a free slot).

Think of it like… a parking lot. Chaining: Each parking spot has a stack of tickets — multiple cars can share a spot (they just line up in the stack).
Open addressing: If your assigned spot is taken, drive to the next empty spot you find.

Strategies Compared

StrategyIdeaTrade-off
ChainingEach bucket holds a linked list of (key, value) pairsSimple; handles high load factors; extra memory for pointers
Linear probingIf bucket i is full, try i+1, i+2, … (wrap around)Better cache performance; clustering can slow lookup
Quadratic probingTry i+1², i+2², i+3², …Reduces clustering compared to linear probing
Double hashingUse a second hash function to compute the probe stepBest distribution; two hash computations per probe

Code Examples

Python 
# Strategy 1: Chaining
class ChainingMap:
    def __init__(self, size=7):
        self.size = size
        self.buckets = [[] for _ in range(size)]

    def _idx(self, key):
        return hash(key) % self.size

    def put(self, key, val):
        bucket = self.buckets[self._idx(key)]
        for i, (k, _) in enumerate(bucket):
            if k == key:
                bucket[i] = (key, val)   # update existing key
                return
        bucket.append((key, val))         # new key → add to chain

    def get(self, key):
        for k, v in self.buckets[self._idx(key)]:
            if k == key:
                return v
        return None


# Strategy 2: Open addressing with linear probing
class LinearProbingMap:
    _DELETED = object()   # sentinel distinguishes "deleted" from "never used"

    def __init__(self, size=11):
        self.size = size
        self.slots = [None] * size   # None = empty; _DELETED = was occupied

    def _probe(self, key):
        """Find the slot for key, or the first empty slot."""
        idx = hash(key) % self.size
        while self.slots[idx] not in (None, self._DELETED):
            if self.slots[idx][0] == key:
                return idx            # found the key
            idx = (idx + 1) % self.size   # probe next slot
        return idx

    def put(self, key, val):
        self.slots[self._probe(key)] = (key, val)

    def get(self, key):
        idx = hash(key) % self.size
        while self.slots[idx] is not None:
            if self.slots[idx] != self._DELETED and self.slots[idx][0] == key:
                return self.slots[idx][1]
            idx = (idx + 1) % self.size
        return None

    def delete(self, key):
        idx = hash(key) % self.size
        while self.slots[idx] is not None:
            if self.slots[idx] != self._DELETED and self.slots[idx][0] == key:
                self.slots[idx] = self._DELETED
                return
            idx = (idx + 1) % self.size


cm = ChainingMap()
cm.put("apple", 1); cm.put("apricot", 2)   # might collide
print(cm.get("apple"))    # 1
print(cm.get("apricot"))  # 2

lp = LinearProbingMap()
lp.put("x", 10); lp.put("y", 20)
print(lp.get("x"))        # 10

// Chaining hashmap
class ChainingMap<V> {
  private buckets: Array<Array<[string, V]>>;
  private size: number;

  constructor(size = 7) {
    this.size = size;
    this.buckets = Array.from({ length: size }, () => []);
  }

  private idx(key: string): number {
    let h = 0;
    for (const ch of key) h = (Math.imul(h, 31) + ch.charCodeAt(0)) % this.size;
    return h;
  }

  put(key: string, val: V): void {
    const b = this.buckets[this.idx(key)];
    const i = b.findIndex(([k]) => k === key);
    if (i >= 0) b[i] = [key, val]; else b.push([key, val]);
  }

  get(key: string): V | undefined {
    return this.buckets[this.idx(key)].find(([k]) => k === key)?.[1];
  }

// Linear probing hashmap
class LinearProbingMap<V> {
  private slots: Array<[string, V] | null | "deleted">;
  private size: number;

  constructor(size = 11) {
    this.size = size;
    this.slots = new Array(size).fill(null);
  }

  private hash(key: string): number {
    let h = 0;
    for (const ch of key) h = (Math.imul(h, 31) + ch.charCodeAt(0)) % this.size;
    return h;
  }

  put(key: string, val: V): void {
    let idx = this.hash(key);
    while (this.slots[idx] !== null && this.slots[idx] !== "deleted") {
      if ((this.slots[idx] as [string, V])[0] === key) {
        this.slots[idx] = [key, val]; return;
      }
      idx = (idx + 1) % this.size;
    }
    this.slots[idx] = [key, val];
  }

  get(key: string): V | undefined {
    let idx = this.hash(key);
    while (this.slots[idx] !== null) {
      const slot = this.slots[idx];
      if (slot !== "deleted" && slot[0] === key) return slot[1];
      idx = (idx + 1) % this.size;
    }
    return undefined;
  }
}

const cm = new ChainingMap<number>();
cm.put("apple", 1); cm.put("apricot", 2);
console.log(cm.get("apple"));    // 1
console.log(cm.get("apricot"));  // 2
Where you see this in real software: Python's dict uses open addressing with pseudo-random probing. Java's HashMap uses chaining. Redis uses chaining with incremental rehashing. Understanding collision resolution helps diagnose why hash tables slow down as they fill up (high load factor).

Bubble Sort

Bubble sort repeatedly steps through the list, compares each pair of adjacent elements, and swaps them if they're in the wrong order. Each full pass "bubbles" the largest remaining value up to its correct position at the end. It's simple to understand but rarely used in production — it's O(n²) in the average and worst case.

Think of it like… bubbles rising through water. The heaviest, biggest value sinks toward the end with every pass, the same way the largest air bubble in a glass always rises to the top first.

Step by Step

  1. Compare items at positions 0 and 1. If the left one is bigger, swap them.
  2. Move to positions 1 and 2, then 2 and 3, and so on to the end of the list — this is one "pass." By the end of a pass, the largest value is guaranteed to be in its final spot.
  3. Repeat the pass, but each time stop one position earlier — the tail is already sorted.
  4. Optimization: if a full pass makes zero swaps, the list is already sorted — stop early instead of running all n passes.

Code Examples

Python 
def bubble_sort(items):
    items = items.copy()
    n = len(items)
    for i in range(n):                    # up to n passes
        swapped = False
        for j in range(0, n - i - 1):      # shrinks by 1 each pass — tail is sorted
            if items[j] > items[j + 1]:
                items[j], items[j + 1] = items[j + 1], items[j]   # swap
                swapped = True
        if not swapped:                    # already sorted — stop early
            break
    return items                           # O(n^2) worst/average, O(n) best case

print(bubble_sort([5, 1, 4, 2, 8]))   # [1, 2, 4, 5, 8]
TypeScript 
function bubbleSort(input: number[]): number[] {
  const items = [...input];
  const n = items.length;
  for (let i = 0; i < n; i++) {              // up to n passes
    let swapped = false;
    for (let j = 0; j < n - i - 1; j++) {    // shrinks by 1 each pass — tail is sorted
      if (items[j] > items[j + 1]) {
        [items[j], items[j + 1]] = [items[j + 1], items[j]];  // swap
        swapped = true;
      }
    }
    if (!swapped) break;                     // already sorted — stop early
  }
  return items;                              // O(n^2) worst/average, O(n) best case
}

console.log(bubbleSort([5, 1, 4, 2, 8]));   // [1, 2, 4, 5, 8]
Where you see this in real software: Almost nowhere in production — it's outclassed by insertion sort at small sizes and by merge/quicksort at large sizes. Its value is entirely educational: the swap-adjacent-elements idea is the easiest sorting mental model to trace by hand.

Selection Sort

Selection sort repeatedly scans the unsorted remainder of the list to find its smallest value, then swaps that value into the next sorted position. Like bubble sort it's O(n²) no matter what — but it makes at most n swaps total, compared to bubble sort's potentially O(n²) swaps, which matters when writes are expensive.

Think of it like… sorting a hand of cards by repeatedly picking the smallest. Scan every card still in your messy pile, pull out the smallest one, and place it at the end of your neat row. Repeat until the messy pile is empty.

Step by Step

  1. Assume the first position holds the "current minimum" search boundary.
  2. Scan the rest of the unsorted portion to find the actual smallest value.
  3. Swap that smallest value into the current boundary position.
  4. Move the boundary one position to the right and repeat until the whole list is scanned.

Code Examples

Python 
def selection_sort(items):
    items = items.copy()
    n = len(items)
    for i in range(n):
        min_idx = i
        for j in range(i + 1, n):          # scan the unsorted remainder
            if items[j] < items[min_idx]:
                min_idx = j
        items[i], items[min_idx] = items[min_idx], items[i]   # one swap per outer step
    return items                           # O(n^2) time, but only O(n) swaps total

print(selection_sort([64, 25, 12, 22, 11]))   # [11, 12, 22, 25, 64]
TypeScript 
function selectionSort(input: number[]): number[] {
  const items = [...input];
  const n = items.length;
  for (let i = 0; i < n; i++) {
    let minIdx = i;
    for (let j = i + 1; j < n; j++) {        // scan the unsorted remainder
      if (items[j] < items[minIdx]) minIdx = j;
    }
    [items[i], items[minIdx]] = [items[minIdx], items[i]];   // one swap per outer step
  }
  return items;                              // O(n^2) time, but only O(n) swaps total
}

console.log(selectionSort([64, 25, 12, 22, 11]));   // [11, 12, 22, 25, 64]
Where you see this in real software: Situations where writes are far more expensive than comparisons — for example, sorting data on flash memory, where minimizing the number of writes extends hardware lifespan. It's also the algorithm behind this site's own sorting playground demo.

Insertion Sort

Insertion sort builds a sorted section one element at a time, taking each new item and sliding it backward into its correct place among the items already sorted. Worst case is O(n²), but it's O(n) on data that's already sorted or nearly sorted — one of the few sorts that gets faster the more ordered its input already is.

Think of it like… sorting playing cards as you pick them up. You hold a sorted hand. Each new card you draw, you slide into its correct spot among the cards you're already holding — you never need to re-sort the whole hand.

Step by Step

  1. Start at the second element — treat the first element as a "sorted" section of size 1.
  2. Save the current element as key, then compare it backward against the sorted section.
  3. Shift every sorted element bigger than key one position to the right, opening a gap.
  4. Drop key into the gap. Move to the next element and repeat.

Code Examples

Python 
def insertion_sort(items):
    items = items.copy()
    for i in range(1, len(items)):
        key = items[i]
        j = i - 1
        while j >= 0 and items[j] > key:   # shift bigger sorted elements right
            items[j + 1] = items[j]
            j -= 1
        items[j + 1] = key                 # drop key into the opened gap
    return items                           # O(n^2) worst case, O(n) if already sorted

print(insertion_sort([12, 11, 13, 5, 6]))   # [5, 6, 11, 12, 13]
TypeScript 
function insertionSort(input: number[]): number[] {
  const items = [...input];
  for (let i = 1; i < items.length; i++) {
    const key = items[i];
    let j = i - 1;
    while (j >= 0 && items[j] > key) {     // shift bigger sorted elements right
      items[j + 1] = items[j];
      j--;
    }
    items[j + 1] = key;                    // drop key into the opened gap
  }
  return items;                            // O(n^2) worst case, O(n) if already sorted
}

console.log(insertionSort([12, 11, 13, 5, 6]));   // [5, 6, 11, 12, 13]
Where you see this in real software: Python's sorted()/list.sort() (Timsort) and Java's Arrays.sort() both fall back to insertion sort for small sub-arrays (typically under ~64 elements) because it beats merge sort's overhead at that size. It's also the natural choice for sorting a live stream of data as it arrives.

Merge Sort

Merge sort splits the list in half recursively until each piece has one element (trivially sorted), then merges pairs of sorted pieces back together in order. It guarantees O(n log n) in every case — best, average, and worst — which is the best possible complexity for a comparison-based sort. The cost is O(n) extra memory for the merge step.

Think of it like… merging two sorted stacks of graded papers. Two teaching assistants each hand you a stack of papers already sorted by name. You merge them into one sorted stack just by repeatedly taking whichever top paper comes first alphabetically — you never have to re-sort either original stack.

Step by Step

  1. Divide: split the list into two halves.
  2. Conquer: recursively sort each half the same way, down to lists of size 0 or 1 (already sorted).
  3. Combine: merge the two sorted halves into one sorted list by repeatedly taking the smaller of the two fronts.

Code Examples

Python 
def merge_sort(items):
    if len(items) <= 1:
        return items
    mid = len(items) // 2
    left = merge_sort(items[:mid])     # recursively sort each half
    right = merge_sort(items[mid:])
    return _merge(left, right)

def _merge(left, right):
    result, i, j = [], 0, 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i]); i += 1
        else:
            result.append(right[j]); j += 1
    result.extend(left[i:])            # append any leftovers
    result.extend(right[j:])
    return result                      # O(n log n) time, O(n) extra space

print(merge_sort([38, 27, 43, 3, 9, 82, 10]))   # [3, 9, 10, 27, 38, 43, 82]
TypeScript 
function mergeSort(items: number[]): number[] {
  if (items.length <= 1) return items;
  const mid = Math.floor(items.length / 2);
  const left = mergeSort(items.slice(0, mid));     // recursively sort each half
  const right = mergeSort(items.slice(mid));
  return merge(left, right);
}

function merge(left: number[], right: number[]): number[] {
  const result: number[] = [];
  let i = 0, j = 0;
  while (i < left.length && j < right.length) {
    if (left[i] <= right[j]) result.push(left[i++]);
    else result.push(right[j++]);
  }
  return [...result, ...left.slice(i), ...right.slice(j)]; // O(n log n) time, O(n) space
}

console.log(mergeSort([38, 27, 43, 3, 9, 82, 10]));   // [3, 9, 10, 27, 38, 43, 82]
Where you see this in real software: External sorting (sorting files too large to fit in memory), and the backbone of Python's and Java's built-in Timsort, which is a hybrid of merge sort and insertion sort. Quick Sort

Quicksort picks a pivot element, partitions the rest of the list into "smaller than pivot" and "bigger than pivot" groups, then recursively sorts each group. It sorts in place (little extra memory) and averages O(n log n) — but a poorly chosen pivot on already-sorted or adversarial data degrades it to O(n²).

Think of it like… sorting people into a line by height, one round at a time. Pick one person as the reference. Everyone shorter steps to their left, everyone taller steps to their right. Now repeat the same trick separately within the "shorter" group and the "taller" group, until every group has one person left.

Step by Step

  1. Choose a pivot: commonly the last element, first element, or a random element.
  2. Partition: rearrange the list so everything less than the pivot comes before it, and everything greater comes after — the pivot lands in its final sorted position.
  3. Recurse: apply the same process independently to the sub-list left of the pivot and the sub-list right of the pivot.
  4. Base case: a sub-list of 0 or 1 elements is already sorted.

Code Examples

Python 
def quick_sort(items):
    if len(items) <= 1:
        return items
    pivot = items[-1]                         # choose the last element as pivot
    smaller = [x for x in items[:-1] if x <= pivot]
    bigger  = [x for x in items[:-1] if x > pivot]
    # sort each side, pivot sits in its final position between them
    return quick_sort(smaller) + [pivot] + quick_sort(bigger)
    # average O(n log n); worst case O(n^2) on adversarial/sorted input + bad pivot choice

print(quick_sort([10, 7, 8, 9, 1, 5]))   # [1, 5, 7, 8, 9, 10]

BST Half-Tree Search

Searching a binary search tree works by using the BST ordering rule to eliminate half the remaining candidates at each step. With one million nodes, you need at most about 20 comparisons (log₂(1,000,000) ≈ 20). This is far more powerful than linear search (which would need up to 1,000,000 comparisons).

Think of it like… "higher or lower?" guessing game. Your target is 42. Middle guess is 50 — too high. Now you only search below 50. Each guess cuts what remains in half. A BST is pre-organized so this halving happens automatically as you walk left or right.

Step by Step (iterative version)

  1. Start at the root node.
  2. Compare the target to the current node's value.
  3. If equal: found! Return true (or the node).
  4. If target is smaller: move to the left child (all smaller values are there).
  5. If target is larger: move to the right child (all larger values are there).
  6. If you reach null: the target is not in the tree. Return false.

def bst_search(root, target):

            return True        # found!
        elif target < node.val:

    return False               # fell off the tree — not found

    return bst_search_recursive(node.right, target)

  return node.val;
}

Heap Self-Reorganization

Every time an element is inserted or removed from a heap, the heap must restore its heap property (the root is always min or max). This is done via two operations: bubble up (also called sift up or heapify up) after insertion, and sift down (heapify down) after removal. Each takes O(log n) because the heap's height is log n.

Think of it like… a bubble rising in water or sinking in oil. A newly inserted small element in a min-heap rises (bubbles up) past larger elements until it finds its correct level. When the min is removed and replaced by the last element, that element sinks down (sifts down) past smaller elements until it stops.

Bubble Up (after insert)

  1. Append the new value to the end of the heap array (last position in the tree).
  2. Compare it with its parent (parent index = ⌊(i-1)/2⌋).
  3. If the new value violates the heap property (e.g., smaller than parent in a min-heap), swap with the parent.
  4. Repeat from the new position until the property holds or you reach the root.

Sift Down (after remove)

  1. Move the last element to the root position to fill the gap left by the removed element.
  2. Compare the root with its two children.
  3. Swap with the smaller child (in a min-heap) if it violates the property.
  4. Repeat from the new position until the property holds or you reach a leaf.

Code Examples

Python 
def heap_push(heap: list, val: int) -> None:
    """Insert into a min-heap and bubble up."""
    heap.append(val)
    i = len(heap) - 1
    while i > 0:
        parent = (i - 1) // 2
        if heap[parent] <= heap[i]:
            break                                         # heap property satisfied
        heap[parent], heap[i] = heap[i], heap[parent]    # swap up
        i = parent


def heap_pop(heap: list) -> int | None:
    """Remove min from a min-heap and sift down the replacement."""
    if not heap:
        return None
    if len(heap) == 1:
        return heap.pop()

    top = heap[0]
    heap[0] = heap.pop()    # move last element to root position
    i, n = 0, len(heap)

    while True:
        left, right = 2 * i + 1, 2 * i + 2
        smallest = i
        if left  < n and heap[left]  < heap[smallest]: smallest = left
        if right < n and heap[right] < heap[smallest]: smallest = right
        if smallest == i:
            break                                              # sift-down complete
        heap[i], heap[smallest] = heap[smallest], heap[i]
        i = smallest

    return top

  heap.push(val);
  let i = heap.length - 1;
  while (i > 0) {
    const parent = Math.floor((i - 1) / 2);

    [heap[parent], heap[i]] = [heap[i], heap[parent]]; // swap up
    i = parent;
  }
}

function heapPop(heap: number[]): number | undefined {  // O(log n)
  if (heap.length === 0) return undefined;
  if (heap.length === 1) return heap.pop();

  const top = heap[0];
  heap[0] = heap.pop()!;  // move last to root
  let i = 0;
  const n = heap.length;

  while (true) {
    const l = 2 * i + 1, r = 2 * i + 2;
    let smallest = i;
    if (l < n && heap[l] < heap[smallest]) smallest = l;
    if (r < n && heap[r] < heap[smallest]) smallest = r;
    if (smallest === i) break;
    [heap[smallest], heap[i]] = [heap[i], heap[smallest]]; // swap down
    i = smallest;
  }
  return top;
}

Depth First Search (DFS)

DFS explores a graph by going as deep as possible along one path before backtracking and trying another. It uses a stack — either explicitly or via recursion (the call stack is a hidden stack). DFS is great for detecting cycles, topological sorting, finding connected components, and solving maze-like problems.

Think of it like… exploring a maze by always taking the first available turn. You go straight ahead until you hit a dead end, then backtrack to the last junction and try the next turn. You eventually visit every room — but not necessarily in the shortest-path order.

Step by Step (recursive)

  1. Mark the current node as visited. Add it to the result.
  2. For each unvisited neighbor: recursively call DFS on that neighbor.
  3. When all neighbors are visited, backtrack (the function returns).
  4. Repeat until no unvisited nodes remain reachable from the start.

# Recursive DFS — O(V + E)
def dfs_recursive(graph, node, visited=None):
    if visited is None:
        visited = set()
    visited.add(node)
    result = [node]
    for neighbor in graph[node]:
        if neighbor not in visited:
            result.extend(dfs_recursive(graph, neighbor, visited))
    return result


# Iterative DFS using an explicit stack — O(V + E)
def dfs_iterative(graph, start):
    visited = set()
    stack = [start]
    result = []
    while stack:
        node = stack.pop()         # LIFO — go deep
        if node in visited:
            continue
        visited.add(node)
        result.append(node)
        # Push neighbors in reverse so leftmost is processed first
        for nb in reversed(graph[node]):
            if nb not in visited:
                stack.append(nb)
    return result


# Cycle detection in a directed graph
def has_cycle(graph, nodes):
    visited = set()
    rec_stack = set()   # nodes currently in the call stack

    def _dfs(node):
        visited.add(node)
        rec_stack.add(node)
        for nb in graph[node]:
            if nb not in visited:
                if _dfs(nb): return True
            elif nb in rec_stack:
                return True          # back edge → cycle!
        rec_stack.remove(node)
        return False

    return any(_dfs(n) for n in nodes if n not in visited)

Breadth First Search (BFS)

BFS explores a graph layer by layer, visiting all nodes at distance 1 from the start, then all at distance 2, and so on. It uses a queue instead of a stack. The key property: BFS always finds the shortest path in an unweighted graph because it never moves further from the start until all closer nodes are visited.

Think of it like… ripples spreading from a stone dropped in water. The first ripple touches nearby points, the second ripple reaches farther out — always in order of increasing distance. BFS is exactly these expanding ripples.

Step by Step

  1. Put the start node in a queue and mark it visited.
  2. While the queue is not empty: dequeue the front node, add it to the result.
  3. For each unvisited neighbor of that node: mark it visited, enqueue it.
  4. Repeat until the queue is empty. Nodes are always processed closest-first.

Code Examples

Python 
from collections import deque, defaultdict


def bfs(graph, start):              # O(V + E)
    visited = {start}
    queue = deque([start])
    order = []
    while queue:
        node = queue.popleft()      # FIFO — process closest node first
        order.append(node)
        for nb in graph[node]:
            if nb not in visited:
                visited.add(nb)
                queue.append(nb)
    return order


def shortest_path(graph, start, end):
    """BFS guarantees the first time we reach 'end', it's via the shortest path."""
    if start == end:
        return [start]
    visited = {start}
    queue = deque([[start]])        # each item in queue is a full path
    while queue:
        path = queue.popleft()
        node = path[-1]
        for nb in graph[node]:
            if nb == end:
                return path + [nb]  # shortest path found!
            if nb not in visited:
                visited.add(nb)
                queue.append(path + [nb])
    return []                       # no path exists

// BFS traversal — O(V + E)
function bfs(graph: Graph, start: string): string[] {
  const visited = new Set([start]);
  const queue: string[] = [start];
  const order: string[] = [];
  while (queue.length) {
    const node = queue.shift()!;    // FIFO
    order.push(node);
    for (const nb of (graph.get(node) ?? [])) {
      if (!visited.has(nb)) { visited.add(nb); queue.push(nb); }
    }
  }
  return order;
}

// Shortest path via BFS — O(V + E)
function shortestPath(graph: Graph, start: string, end: string): string[] {
  if (start === end) return [start];
  const visited = new Set([start]);
  const queue: string[][] = [[start]];
  while (queue.length) {
    const path = queue.shift()!;
    const node = path[path.length - 1];
    for (const nb of (graph.get(node) ?? [])) {
      if (nb === end) return [...path, nb];
      if (!visited.has(nb)) { visited.add(nb); queue.push([...path, nb]); }
    }
  }
  return [];
}

Trie Prefix Search

Trie prefix search finds every word stored in a trie that starts with a given prefix. It works in two phases: first navigate to the node representing the last character of the prefix (O(L) where L = prefix length), then collect all words in the subtree rooted there. This is the algorithm behind "Google Suggest" and every autocomplete field.

Think of it like… navigating a phone book by index tabs. You flip to the "SM" tab instantly. From there, you read every entry: Smith, Smythe, Smithson… You didn't scan from A — you jumped to the right starting point first.

Step by Step

  1. Start at the root. Set current node = root.
  2. For each character in the prefix: if the child for that character exists, move to it. If any character is missing, return an empty list (no words share this prefix).
  3. You are now at the prefix node. Collect all words in its subtree using DFS.
  4. At each node: if it's marked end-of-word, add the accumulated prefix to results. Continue into each child, appending that child's character.

class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False

class AutocompleteTrie:
    def __init__(self):
        self.root = TrieNode()

    def insert(self, word):              # O(L) — L = word length
        node = self.root
        for ch in word:
            node.children.setdefault(ch, TrieNode())
            node = node.children[ch]
        node.is_end = True

    def _navigate_to(self, prefix):
        """Walk to the node for the last character of prefix. O(L)"""
        node = self.root
        for ch in prefix:
            if ch not in node.children:
                return None   # prefix not in trie
            node = node.children[ch]
        return node

    def _collect_all(self, node, current_prefix):
        """DFS: collect every word in the subtree. O(results)"""
        results = []
        if node.is_end:
            results.append(current_prefix)
        for ch, child in node.children.items():
            results.extend(self._collect_all(child, current_prefix + ch))
        return results

    def autocomplete(self, prefix):      # O(L + results)
        node = self._navigate_to(prefix)
        if node is None:
            return []
        return self._collect_all(node, prefix)

    def count_words_with_prefix(self, prefix):
        return len(self.autocomplete(prefix))

    for (const ch of word) {
      if (!node.children.has(ch)) node.children.set(ch, new TrieNode());
      node = node.children.get(ch)!;
    }
    node.isEnd = true;

    let node = this.root;
    for (const ch of prefix) {
      if (!node.children.has(ch)) return null;
      node = node.children.get(ch)!;
    }
    return node;

    const results: string[] = [];
    if (node.isEnd) results.push(prefix);
    for (const [ch, child] of node.children) {
      results.push(...this.collectAll(child, prefix + ch));
    }
    return results;
  }

  autocomplete(prefix: string): string[] {  // O(L + results)
    const node = this.navigateTo(prefix);
    return node ? this.collectAll(node, prefix) : [];
  }
}

Path Compression

Path compression is an optimization in the Union Find (Disjoint Set) data structure. Without it, the find() operation could walk a long chain of parent pointers (O(n) worst case). Path compression collapses those chains: after finding the root, every node visited along the way is pointed directly to the root. Future lookups for those nodes become O(1).

Think of it like… updating your home address after moving. Without path compression: every time someone visits you, they follow a chain of forwarding addresses (old address → old address → current address). With path compression: after following the chain once, everyone's forwarding address is updated to point directly to your current home. The next visit is instant.

Step by Step

  1. Call find(x) to locate the root of x's group.
  2. While walking up the parent chain to reach the root, record every node visited.
  3. After finding the root, set parent[node] = root for every node on the path.
  4. On subsequent calls, those nodes return their root in one step.

Code Examples

Python 
parent = list(range(8))   # 8 nodes, each initially its own root

# WITHOUT path compression — O(n) worst case for a long chain
def find_naive(x):
    while parent[x] != x:
        x = parent[x]      # walk up one step at a time
    return x

# WITH path compression (recursive) — O(α(n)) amortized
def find(x):
    if parent[x] != x:
        parent[x] = find(parent[x])   # flatten: point directly to root
    return parent[x]

# WITH path compression (iterative halving — slightly different strategy)

TypeScript 
class UnionFind {
  private parent: number[];
  private rank: number[];

  constructor(n: number) {
    this.parent = Array.from({ length: n }, (_, i) => i);
    this.rank   = new Array(n).fill(0);
  }

  // find() with recursive path compression — O(α(n)) amortized
  find(x: number): number {
    if (this.parent[x] !== x) {
      this.parent[x] = this.find(this.parent[x]); // compress path
    }
    return this.parent[x];
  }

  // Iterative path halving (alternative compression — same amortized cost)
  findHalving(x: number): number {
    while (this.parent[x] !== x) {
      this.parent[x] = this.parent[this.parent[x]]; // skip a level
      x = this.parent[x];
    }
    return x;
  }

  union(x: number, y: number): void {
    const rx = this.find(x), ry = this.find(y);
    if (rx === ry) return;
    if (this.rank[rx] < this.rank[ry]) this.parent[rx] = ry;
    else if (this.rank[rx] > this.rank[ry]) this.parent[ry] = rx;
    else { this.parent[ry] = rx; this.rank[rx]++; }
  }

  connected(x: number, y: number): boolean {
    return this.find(x) === this.find(y);
  }
}

const uf = new UnionFind(8);
// Build chain: 5 → 3 → 1 → 0
uf.union(0, 1); uf.union(1, 3); uf.union(3, 5);

console.log(uf.find(5));           // 0 (root)
// After find(5), parent[5], parent[3], parent[1] all point directly to 0
console.log(uf.connected(5, 0));   // true
console.log(uf.connected(5, 2));   // false

LRU Eviction

LRU (Least Recently Used) eviction decides which cached item to discard when the cache is full. The item that has not been accessed for the longest time is evicted. The algorithm maintains a recency order — most recently used at the front, least recently used at the back. On every access, the item moves to the front. On every eviction, the back item is removed.

Think of it like… a row of lockers sorted by last use. Every time you access your locker, it moves to the front of the row. When a new student needs a locker and there are none left, the locker at the very back — the one nobody has touched in the longest time — gets cleared out.

Why Hashmap + Doubly Linked List?

NeedSolved by
O(1) lookup by keyHashmap: key → pointer to node in list
O(1) move-to-front on accessDoubly linked list: pointer lets you splice any node out and reinsert at head instantly
O(1) evict from backDoubly linked list tail is always the LRU

from collections import OrderedDict

class LRUCache:
    """
    Python's OrderedDict is a dict backed by a doubly linked list,
    making it a natural LRU cache. All operations are O(1).
    """
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = OrderedDict()

    def get(self, key: str) -> int:
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)   # mark as most recently used
        return self.cache[key]

    def put(self, key: str, value: int) -> None:
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.capacity:
            evicted_key, _ = self.cache.popitem(last=False)   # evict LRU
            print(f"  Evicted: {evicted_key}")


# Trace through a cache with capacity 3
cache = LRUCache(3)
cache.put("a", 1)   # cache: a
cache.put("b", 2)   # cache: a, b
cache.put("c", 3)   # cache: a, b, c

print(cache.get("a"))   # 1   — still in cache (was recently accessed)
print(cache.get("d"))   # 4

// Doubly linked list node for the LRU cache
class DLLNode {
  prev: DLLNode | null = null;
  next: DLLNode | null = null;
  constructor(public key: string, public val: number) {}
}

class LRUCache {
  private map = new Map<string, DLLNode>(); // key → node
  // Sentinel head (most recent) and tail (least recent) — simplify edge cases
  private head = new DLLNode("", 0);
  private tail = new DLLNode("", 0);

  constructor(private capacity: number) {
    this.head.next = this.tail;
    this.tail.prev = this.head;
  }

  private insertFront(node: DLLNode): void {  // O(1)
    node.next = this.head.next;
    node.prev = this.head;
    this.head.next!.prev = node;
    this.head.next = node;
  }

  private remove(node: DLLNode): void {       // O(1)
    node.prev!.next = node.next;
    node.next!.prev = node.prev;
  }

  get(key: string): number {                  // O(1)
    const node = this.map.get(key);
    if (!node) return -1;
    this.remove(node);
    this.insertFront(node);                   // move to front (most recent)
    return node.val;
  }

  put(key: string, val: number): void {       // O(1)
    if (this.map.has(key)) this.remove(this.map.get(key)!);
    const node = new DLLNode(key, val);
    this.insertFront(node);
    this.map.set(key, node);
    if (this.map.size > this.capacity) {
      const lru = this.tail.prev!;            // LRU is just before sentinel tail
      this.remove(lru);
      this.map.delete(lru.key);
    }
  }
}

const cache = new LRUCache(3);
cache.put("a", 1); cache.put("b", 2); cache.put("c", 3);
cache.get("a");         // "a" is now most recent
cache.put("d", 4);      // "b" evicted

console.log(cache.get("b")); // -1  — evicted
console.log(cache.get("a")); // 1   — still in cache
console.log(cache.get("d")); // 4
